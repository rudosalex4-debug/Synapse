cmake_minimum_required(VERSION 3.22)
project(synapse VERSION 0.12.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAXHELP_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Drogon CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(PostgreSQL REQUIRED)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
add_library(maxhelp_domain src/domain.cpp src/max_bot.cpp src/catalog.cpp src/matching.cpp)
target_include_directories(maxhelp_domain PUBLIC include)
target_link_libraries(maxhelp_domain PUBLIC OpenSSL::Crypto CURL::libcurl nlohmann_json::nlohmann_json)
add_library(maxhelp_core src/config.cpp src/db.cpp src/services.cpp src/jobs.cpp src/workflow.cpp src/maintenance.cpp src/notifications.cpp src/bot_workflow.cpp src/moderation.cpp src/community_rules.cpp)
target_link_libraries(maxhelp_core PUBLIC maxhelp_domain PostgreSQL::PostgreSQL Threads::Threads)
add_executable(max-help src/main.cpp src/server.cpp)
target_link_libraries(max-help PRIVATE maxhelp_core Drogon::Drogon)
foreach(target maxhelp_domain maxhelp_core max-help)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
 if(MAXHELP_SANITIZERS)
  target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
add_executable(domain-tests tests/domain_test.cpp)
target_link_libraries(domain-tests PRIVATE maxhelp_domain)
add_test(NAME domain-tests COMMAND domain-tests)
add_executable(max-bot-tests tests/max_bot_test.cpp)
target_link_libraries(max-bot-tests PRIVATE maxhelp_domain)
add_test(NAME max-bot-tests COMMAND max-bot-tests)
# Separate opt-in database suite: migrated disposable DB, no running worker.
add_executable(queue-tests tests/queue_test.cpp)
target_link_libraries(queue-tests PRIVATE maxhelp_core)
add_executable(bot-workflow-tests tests/bot_workflow_test.cpp)
target_link_libraries(bot-workflow-tests PRIVATE maxhelp_core)
target_compile_definitions(bot-workflow-tests PRIVATE SYNAPSE_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../data")

add_executable(catalog-tests tests/catalog_test.cpp)
target_link_libraries(catalog-tests PRIVATE maxhelp_domain)
add_test(NAME catalog-tests COMMAND catalog-tests)

add_executable(matching-tests tests/matching_test.cpp)
target_link_libraries(matching-tests PRIVATE maxhelp_domain)
target_compile_definitions(matching-tests PRIVATE SYNAPSE_CATALOG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../data/taxonomy.example.json")
add_test(NAME matching-tests COMMAND matching-tests)
